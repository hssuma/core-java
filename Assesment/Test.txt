1) Expalin Constructor chaining using this() with example?
	Constructor chaining can be achive dusing this() statement
	Syntax
	ConstructorName()
	{
		this();
	}
	
	
	this(); statement must be the first line to achive the constructor chaining
	
	
	
	class Amazon
	{
		String product;
		double cost;
		Amazon()
		{
			this("Smart Watch");
			System.out.println("This is the default constructor");
		}
		Amazon(String product)
		{
			this("Smart Watch", 2500.45);
			this.product = product;
			System.out.println("This is the parameterized constructor");
		}
		
		Amazon(String product, double cost)
		{
			this.product = product;
			this.cost = cost;
		}
		
		void getDeatils()
		{
			System.out.println("The product is "+product);
			System.out.println("The cost of the product is "+cost);
		}
	
	}
	
	
	
	
	
	class AmazonExec
	{
		public static void main(String args[])
		{
			Amazon a = new Amazon();
			a.getDeatils();
		}
	}

5) Write a program using parameterized constructor



	parameterized constructor is a constructor with some values or arguments
	
	
	class Factory
	{
		int factoryId;
		String factoryName;
		
		Factory(int id)
		{
			factoryId = id;
			System.out.println("This is the parameterized constructor with the parameter called id");
			System.out.println("The factoryId is "+factoryId);
		}
		
		Factory(String name)
		{
			factoryName = name;
			System.out.println("This is the parameterized constructor with parameter called name");
			System.out.println("The factoryName is "+factoryName);
		
		}
		
		Factory(int factoryId, String factoryName)
		{
			this.factoryId  = factoryId;
			this.factoryName = factoryName;
			System.out.println("the factoryId is "+factoryId);
			System.out.println("The factoryName is "+factoryName);
		}
	}
	
	
	
	
	
	
	
	class FactoryExec
	{
	
		public static void main(String f[])
		{
			Factory f = new Factory(1);
			Factory fTwo = new Factory("XYZ");
			Factory fThree = new Factory(2, "ABC");
		}
	}
